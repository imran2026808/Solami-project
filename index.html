const connection = new solanaWeb3.Connection(SOLAMI_RPC, 'confirmed'); async function updateSlot() { try { const slot = await connection.getSlot(); document.getElementById('slot-value').innerText = slot; } catch (error) { console.error("Error fetching slot:", error); document.getElementById('network-status').innerText = "RPC Error"; document.getElementById('network-status').className = "stat online error"; } } // Balance Check Function document.getElementById('check-btn').addEventListener('click', async () => { const inputAddress = document.getElementById('wallet-input').value.trim(); const resultBox = document.getElementById('balance-result'); try { resultBox.innerText = "Fetching balance..."; const publicKey = new solanaWeb3.PublicKey(inputAddress); const balance = await connection.getBalance(publicKey); const solBalance = balance / solanaWeb3.LAMPORTS_PER_SOL; resultBox.innerText = Balance: ${solBalance} SOL; } catch (error) { resultBox.innerText = "Error: Invalid wallet address or network issue."; } }); // हर 5 सेकंड में स्लॉट अपडेट करें updateSlot(); setInterval(updateSlot, 5000);
